### Task 2: Create pricing utility module

**Files:**
- Create: `lib/pricing.ts`

- [ ] **Step 1: Create `lib/pricing.ts` with all pricing functions**

```typescript
import type { Meal } from "./types";

export const STANDARD_DISCOUNT = 0.10;

export function getAveragePrice(meals: Meal[]): number {
  if (meals.length === 0) return 0;
  const total = meals.reduce((sum, meal) => sum + meal.price, 0);
  return total / meals.length;
}

export function getStandardWeeklyPrice(
  avgPrice: number,
  mealsPerDay: number,
  daysPerWeek: number
): number {
  const totalMeals = mealsPerDay * daysPerWeek;
  return avgPrice * totalMeals * (1 - STANDARD_DISCOUNT);
}

export function getFlexibleWeeklyPrice(
  avgPrice: number,
  mealsPerDay: number,
  daysPerWeek: number
): number {
  const totalMeals = mealsPerDay * daysPerWeek;
  return avgPrice * totalMeals;
}

export function getSelectedMealsTotal(
  meals: Meal[],
  selected: Record<string, number>
): number {
  return Object.entries(selected).reduce((sum, [mealId, qty]) => {
    const meal = meals.find((m) => m.id === mealId);
    return sum + (meal?.price ?? 0) * qty;
  }, 0);
}

export function formatPrice(amount: number): string {
  return `€${amount.toFixed(2)}`;
}
```

Two approaches: (a) make a server wrapper that fetches and passes props, or (b) fetch client-side. Since this page is already `"use client"`, the simplest approach is to fetch the average price from a small server component wrapper or use `useEffect` to fetch.

The cleanest approach: create a server action or API route that returns the average price. But the simplest approach matching the existing pattern: convert to a server/client split like the menu page. However, the wizard has heavy client state, so we'll fetch meals in a parent server component and pass just the average price down.
